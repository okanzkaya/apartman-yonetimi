CREATE PROCEDURE sp_AktifArizalariListele
AS
BEGIN
    SELECT 
        A.ArizaID,
        K.Ad + ' ' + K.Soyad AS BildirenKisi,
        D.Blok + ' Blok, No: ' + CAST(D.KapiNo AS NVARCHAR) AS DaireBilgisi,
        A.Kategori,
        A.Baslik,
        A.BildirimTarihi,
        A.Durum
    FROM 
        ArizaBildirimleri A
    INNER JOIN 
        Kullanicilar K ON A.BildirenKullaniciID = K.KullaniciID
    LEFT JOIN 
        Daireler D ON K.DaireID = D.DaireID
    WHERE 
        A.Durum != 'Çözüldü'
    ORDER BY 
        A.BildirimTarihi DESC;
END;
GO
